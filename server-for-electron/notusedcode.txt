        export function count_data(type) {
    let rowCount = 0;

    fs.createReadStream('cdr/' + type + '.csv')
        .pipe(csv())
        .on('data', (row) => { rowCount++; })
        .on('end', () => {
            if (type === "Internal")
                cdr_count[0] = rowCount;
            else if (type === "Incoming")
                cdr_count[1] = rowCount;
            else if (type === "Outgoing")
                cdr_count[2] = rowCount;
        })
        .on('error', (err) => { console.error(err); });
}
